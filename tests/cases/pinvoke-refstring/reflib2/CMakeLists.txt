cmake_minimum_required(VERSION 3.10)
project(reflib2 CXX)
add_library(reflib2 STATIC reflib2.cpp)
target_link_libraries(reflib2 PUBLIC ole32)
target_compile_features(reflib2 PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(reflib2 PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(reflib2 PRIVATE -Wall -Wextra)
endif()
